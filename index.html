function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("DailyLogs") || SpreadsheetApp.getActiveSpreadsheet().insertSheet("DailyLogs");
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "Date", "Student_ID", "Student_Name", "Optional", "Status", "GS_Hrs", "Opt_Hrs", "Mains_Hrs", "Mains_Qs", "Prelims_Hrs", "Prelims_MCQs", "CSAT_Hrs", "Total_Hrs", "Source", "Locked", "Remarks"]);
  }
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), data.date, data.id, data.name, data.optional, data.status, data.gs, data.opt, data.mainsHrs, data.mainsQs, data.prelimsHrs, data.prelimsMCQs, data.csat, data.total, data.source, data.locked, data.remarks]);
  PropertiesService.getScriptProperties().setProperty("STATE_JSON", data.fullStateJson || "{}");
  return ContentService.createTextOutput(JSON.stringify({status: "ok"})).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var state = PropertiesService.getScriptProperties().getProperty("STATE_JSON") || "{}";
  return ContentService.createTextOutput(state).setMimeType(ContentService.MimeType.JSON);
}
